document.addEventListener("DOMContentLoaded", () => { const canvas = document.getElementById("physicsCanvas"); if (!canvas || typeof THREE === "undefined") { return; } // ========================= // Scene // ========================= const scene = new THREE.Scene(); scene.background = new THREE.Color(0x07111f); // ========================= // Camera // ========================= const camera = new THREE.PerspectiveCamera( 45, 1, 0.1, 1000 ); camera.position.set(0, 7, 15); // ========================= // Renderer // ========================= const renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true, alpha: false }); renderer.setPixelRatio( Math.min(window.devicePixelRatio || 1, 2) ); renderer.outputColorSpace = THREE.SRGBColorSpace; // ========================= // Lights // ========================= const ambientLight = new THREE.AmbientLight( 0xffffff, 1.4 ); scene.add(ambientLight); const pointLight = new THREE.PointLight( 0xffffff, 2.5, 100 ); pointLight.position.set(0, 5, 5); scene.add(pointLight); // ========================= // Main Group // ========================= const systemGroup = new THREE.Group(); scene.add(systemGroup); // ========================= // Central Planet // ========================= const planetGeometry = new THREE.SphereGeometry(2.1, 48, 48); const planetMaterial = new THREE.MeshStandardMaterial({ color: 0x2277ff, roughness: 0.55, metalness: 0.15 }); const planet = new THREE.Mesh( planetGeometry, planetMaterial ); systemGroup.add(planet); // ========================= // Planet Glow // ========================= const glowGeometry = new THREE.SphereGeometry(2.35, 32, 32); const glowMaterial = new THREE.MeshBasicMaterial({ color: 0x3388ff, transparent: true, opacity: 0.12, side: THREE.BackSide }); const glow = new THREE.Mesh( glowGeometry, glowMaterial ); systemGroup.add(glow); // ========================= // Orbit // ========================= const orbitRadius = 5.5; const orbitCurve = new THREE.EllipseCurve( 0, 0, orbitRadius, orbitRadius, 0, Math.PI * 2, false, 0 ); const orbitPoints = orbitCurve.getPoints(128); const orbitGeometry = new THREE.BufferGeometry().setFromPoints( orbitPoints.map( point => new THREE.Vector3( point.x, 0, point.y ) ) ); const orbitMaterial = new THREE.LineBasicMaterial({ color: 0x66bbff, transparent: true, opacity: 0.55 }); const orbit = new THREE.LineLoop( orbitGeometry, orbitMaterial ); systemGroup.add(orbit); // ========================= // Satellite // ========================= const satelliteGroup = new THREE.Group(); systemGroup.add(satelliteGroup); const satelliteGeometry = new THREE.SphereGeometry( 0.65, 32, 32 ); const satelliteMaterial = new THREE.MeshStandardMaterial({ color: 0xffb52e, roughness: 0.4, metalness: 0.2 }); const satellite = new THREE.Mesh( satelliteGeometry, satelliteMaterial ); satellite.position.x = orbitRadius; satelliteGroup.add(satellite); // ========================= // Satellite Ring // ========================= const ringGeometry = new THREE.TorusGeometry( 0.95, 0.06, 12, 48 ); const ringMaterial = new THREE.MeshBasicMaterial({ color: 0xffdd66 }); const satelliteRing = new THREE.Mesh( ringGeometry, ringMaterial ); satelliteRing.rotation.x = Math.PI / 2; satellite.add(satelliteRing); // ========================= // Direction Indicator // ========================= const arrowDirection = new THREE.Vector3(1, 0, 0); const arrowOrigin = new THREE.Vector3( orbitRadius, 0, 0 ); const arrow = new THREE.ArrowHelper( arrowDirection, arrowOrigin, 2, 0xffffff, 0.45, 0.25 ); satelliteGroup.add(arrow); // ========================= // Stars // ========================= const starsGeometry = new THREE.BufferGeometry(); const starPositions = []; for (let i = 0; i < 500; i++) { const x = (Math.random() - 0.5) * 80; const y = (Math.random() - 0.5) * 60; const z = (Math.random() - 0.5) * 80; starPositions.push(x, y, z); } starsGeometry.setAttribute( "position", new THREE.Float32BufferAttribute( starPositions, 3 ) ); const starsMaterial = new THREE.PointsMaterial({ color: 0xffffff, size: 0.08, transparent: true, opacity: 0.8 }); const stars = new THREE.Points( starsGeometry, starsMaterial ); scene.add(stars); // ========================= // Controls // ========================= let autoRotate = true; let isDragging = false; let previousX = 0; let previousY = 0; let rotationX = 0; let rotationY = 0; // ========================= // Pointer Down // ========================= canvas.addEventListener( "pointerdown", event => { isDragging = true; previousX = event.clientX; previousY = event.clientY; canvas.setPointerCapture( event.pointerId ); } ); // ========================= // Pointer Move // ========================= canvas.addEventListener( "pointermove", event => { if (!isDragging) { return; } const deltaX = event.clientX - previousX; const deltaY = event.clientY - previousY; rotationY += deltaX * 0.01; rotationX += deltaY * 0.01; rotationX = Math.max( -1.2, Math.min(1.2, rotationX) ); systemGroup.rotation.y = rotationY; systemGroup.rotation.x = rotationX; previousX = event.clientX; previousY = event.clientY; } ); // ========================= // Pointer Up // ========================= canvas.addEventListener( "pointerup", event => { isDragging = false; try { canvas.releasePointerCapture( event.pointerId ); } catch (error) {} } ); canvas.addEventListener( "pointercancel", () => { isDragging = false; } ); // ========================= // Zoom // ========================= canvas.addEventListener( "wheel", event => { event.preventDefault(); camera.position.z += event.deltaY * 0.01; camera.position.z = Math.max( 7, Math.min( 25, camera.position.z ) ); }, { passive: false } ); // ========================= // Reset // ========================= const resetButton = document.getElementById( "physicsReset" ); if (resetButton) { resetButton.addEventListener( "click", () => { rotationX = 0; rotationY = 0; systemGroup.rotation.set( 0, 0, 0 ); camera.position.set( 0, 7, 15 ); } ); } // ========================= // Rotation Toggle // ========================= const rotateButton = document.getElementById( "physicsRotateToggle" ); if (rotateButton) { rotateButton.addEventListener( "click", () => { autoRotate = !autoRotate; rotateButton.textContent = autoRotate ? "⏸ إيقاف الدوران" : "▶ تشغيل الدوران"; } ); } // ========================= // Resize // ========================= function resizeRenderer() { const width = canvas.clientWidth || 800; const height = canvas.clientHeight || 500; renderer.setSize( width, height, false ); camera.aspect = width / height; camera.updateProjectionMatrix(); } window.addEventListener( "resize", resizeRenderer ); resizeRenderer(); // ========================= // Animation // ========================= let previousTime = performance.now(); function animate(currentTime) { requestAnimationFrame( animate ); const delta = Math.min( (currentTime - previousTime) / 1000, 0.05 ); previousTime = currentTime; // حركة القمر حول الجسم المركزي satelliteGroup.rotation.y += delta * 0.65; // دوران الكوكب planet.rotation.y += delta * 0.25; // دوران الحلقة satelliteRing.rotation.z += delta * 0.8; // دوران تلقائي للمشهد if (autoRotate && !isDragging) { systemGroup.rotation.y += delta * 0.12; } stars.rotation.y += delta * 0.003; renderer.render( scene, camera ); } animate( performance.now() ); });
